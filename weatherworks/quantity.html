// quantity.html 에서 물량 추가 및 작업일수 자동 계산 함수 예시 function addQuantityToProject() { const db = loadDB(); const pid = getCurrentProjectId(db); // 상단에서 선택된 현장 ID (p1, p2 등) const workType = document.getElementById('workTypeSelect').value; // 예: 토공사 const itemSpec = document.getElementById('itemSpecName').textContent; // 예: 되메우기 및 다짐 const totalQty = parseFloat(document.getElementById('totalQtyInput').value); // 입력한 총 물량 const dailyRate = parseFloat(document.getElementById('dailyRateValue').textContent); // 표준 일일작업량 if (!totalQty || totalQty <= 0) { alert("총 물량을 입력해 주세요."); return; } // 1. 순수 작업일수 계산 (총 물량 ÷ 일일 작업량) const calcDays = Math.ceil(totalQty / dailyRate); // 2. 해당 현장(pid)의 공정 목록 불러오기 let projectTaskMap = JSON.parse(localStorage.getItem('my_project_tasks_map') || '{}'); if (!projectTaskMap[pid]) { projectTaskMap[pid] = []; } // 3. 기존 등록된 마지막 공정의 종료일 이후로 자동 시작일 배치 const currentTasks = projectTaskMap[pid]; const project = db.projects.find(x => x.id === pid); let startDate = project ? project.startDate : new Date().toISOString().slice(0, 10); if (currentTasks.length > 0) { const lastTask = currentTasks[currentTasks.length - 1]; const nextStart = new Date(new Date(lastTask.end).getTime() + 86400000); startDate = nextStart.toISOString().slice(0, 10); } const endDate = new Date(new Date(startDate).getTime() + (calcDays - 1) * 86400000).toISOString().slice(0, 10); // 4. 현장 공정 데이터로 추가 projectTaskMap[pid].push({ id: Date.now(), name: `${workType} - ${itemSpec}`, qty: totalQty, dailyRate: dailyRate, days: calcDays, start: startDate, end: endDate, progress: 0 // 신규 등록 시 진행률 0% }); localStorage.setItem('my_project_tasks_map', JSON.stringify(projectTaskMap)); alert(`[${itemSpec}] 소요 작업일수 ${calcDays}일이 해당 현장 공정표에 반영되었습니다.`); }
